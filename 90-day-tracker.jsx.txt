/* ---------- helpers ---------- */

const phaseOf = (n) => PHASES.find((p) => n >= p.from && n <= p.to);
const weekOf = (n) => WEEKS.find((w) => n >= w.from && n <= w.to);
const emptyDay = () => ({ checks: Array(9).fill(false), win: "", won: "", drift: "", tomorrow: "" });
const count = (d) => (d ? d.checks.filter(Boolean).length : 0);

function todayNumber() {
  const now = new Date();
  const t0 = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  return Math.round((t0 - START) / 86400000) + 1; // may be <1 or >90
}

const STORAGE_KEY = "plan90-v1";

/* ---------- small components ---------- */

function Field({ label, value, onChange, rows = 2, placeholder }) {
  return (
    <label className="field">
      <span className="field-label">{label}</span>
      <textarea rows={rows} value={value || ""} placeholder={placeholder || "Write here"} onChange={(e) => onChange(e.target.value)} />
    </label>
  );
}

function CheckRow({ label, checked, onToggle, color }) {
  return (
    <button type="button" className={"check" + (checked ? " on" : "")} style={{ "--c": color }} onClick={onToggle} role="checkbox" aria-checked={checked}>
      <span className="box" aria-hidden="true">{checked ? "✓" : ""}</span>
      <span className="check-text">{label}</span>
    </button>
  );
}

/* ---------- main ---------- */

export default function Tracker() {
  const todayN = todayNumber();
  const todayClamped = Math.min(90, Math.max(1, todayN));

  const [data, setData] = useState({ days: {}, weeks: {}, plan: { targets: {}, skill: "", output: "", drift: ["", "", ""], phases: {}, phaseChecks: {} } });
  const [loaded, setLoaded] = useState(false);
  const [saveState, setSaveState] = useState("idle"); // idle | saving | saved | unavailable
  const [view, setView] = useState("day");
  const [sel, setSel] = useState(todayClamped);
  const timer = useRef(null);
  const didLoad = useRef(false);

  useEffect(() => {
    (async () => {
      try {
        const r = await window.storage.get(STORAGE_KEY);
        if (r && r.value) {
          const parsed = JSON.parse(r.value);
          setData((d) => ({ ...d, ...parsed, plan: { ...d.plan, ...(parsed.plan || {}) } }));
        }
      } catch (e) {
        /* first visit: nothing saved yet */
      }
      didLoad.current = true;
      setLoaded(true);
    })();
  }, []);

  useEffect(() => {
    if (!didLoad.current) return;
    setSaveState("saving");
    clearTimeout(timer.current);
    timer.current = setTimeout(async () => {
      try {
        const r = await window.storage.set(STORAGE_KEY, JSON.stringify(data));
        setSaveState(r ? "saved" : "unavailable");
      } catch (e) {
        setSaveState("unavailable");
      }
    }, 500);
    return () => clearTimeout(timer.current);
  }, [data]);

  /* updaters */
  const getDay = (n) => data.days[n] || emptyDay();
  const patchDay = (n, patch) => setData((d) => ({ ...d, days: { ...d.days, [n]: { ...(d.days[n] || emptyDay()), ...patch } } }));
  const toggleDay = (n, i) => {
    const cur = getDay(n);
    const checks = cur.checks.map((v, k) => (k === i ? !v : v));
    patchDay(n, { checks });
  };
  const getWeek = (n) => data.weeks[n] || { checks: Array(5).fill(false), top: ["", "", ""], review: Array(5).fill("") };
  const patchWeek = (n, patch) => setData((d) => ({ ...d, weeks: { ...d.weeks, [n]: { ...(d.weeks[n] || { checks: Array(5).fill(false), top: ["", "", ""], review: Array(5).fill("") }), ...patch } } }));
  const patchPlan = (patch) => setData((d) => ({ ...d, plan: { ...d.plan, ...patch } }));
  const setIn = (arr, i, v) => arr.map((x, k) => (k === i ? v : x));

  /* stats */
  const stats = useMemo(() => {
    let full = 0, partial = 0;
    for (let n = 1; n <= 90; n++) {
      const c = count(data.days[n]);
      if (c === 9) full++;
      else if (c > 0) partial++;
    }
    let streak = 0;
    let n = Math.min(90, Math.max(0, todayN));
    if (n >= 1 && count(data.days[n]) < 9) n -= 1; // today may still be in progress
    while (n >= 1 && count(data.days[n]) === 9) { streak++; n--; }
    return { full, partial, streak };
  }, [data.days, todayN]);

  const phase = phaseOf(sel);
  const week = weekOf(sel);
  const dayData = getDay(sel);
  const dayCount = count(dayData);
  const yesterdayMissed = sel === todayN && sel > 1 && count(data.days[sel - 1]) === 0;

  const cellStatus = (n) => {
    const c = count(data.days[n]);
    if (c === 9) return "full";
    if (c > 0) return "part";
    if (n < todayN) return "miss";
    return "future";
  };

  const goDay = (n) => { setSel(n); setView("day"); window.scrollTo?.({ top: 0 }); };

  return (
    <div className="app">
      <style>{CSS}</style>

      {/* ---------- Top: overall check view ---------- */}
      <header className="top">
        <div className="title-row">
          <div>
            <h1>90-Day Back on Track</h1>
            <p className="sub">Oct 1 to Dec 29, 2026. Return, rebuild, rise.</p>
          </div>
          <div className="save" data-state={saveState}>
            {!loaded ? "Loading" : saveState === "saving" ? "Saving…" : saveState === "saved" ? "Saved" : saveState === "unavailable" ? "Not saving" : "Ready"}
          </div>
        </div>

        <div className="stats">
          <div><b>{todayN < 1 ? "Starts Oct 1" : todayN > 90 ? "Finished" : `Day ${todayN}`}</b><span>of 90</span></div>
          <div><b>{stats.full}</b><span>full days</span></div>
          <div><b>{stats.streak}</b><span>day streak</span></div>
          <div><b>{stats.partial}</b><span>partial days</span></div>
        </div>

        <div className="grid-wrap">
          {PHASES.map((p) => (
            <div className="phase-row" key={p.id}>
              <div className="phase-tag" style={{ color: p.color }}>
                <b>{p.name}</b>
                <span>{p.range}</span>
              </div>
              <div className="cells">
                {Array.from({ length: 30 }, (_, i) => p.from + i).map((n) => {
                  const st = cellStatus(n);
                  const c = count(data.days[n]);
                  const op = st === "full" ? 1 : st === "part" ? 0.22 + 0.55 * (c / 9) : 1;
                  return (
                    <button
                      key={n}
                      type="button"
                      className={`cell ${st}${n === todayN ? " today" : ""}${n === sel ? " sel" : ""}`}
                      style={{ "--c": p.color, "--o": op }}
                      onClick={() => goDay(n)}
                      aria-label={`Day ${n}, ${fmtShort(dateOf(n))}, ${c} of 9 done`}
                    >
                      <i />
                      <em>{st === "full" ? "✓" : n}</em>
                    </button>
                  );
                })}
              </div>
            </div>
          ))}
          <div className="legend">
            <span><i className="lg full" />All 9 done</span>
            <span><i className="lg part" />Some done</span>
            <span><i className="lg miss" />Missed</span>
            <span><i className="lg future" />Ahead</span>
          </div>
        </div>

        <nav className="tabs" role="tablist">
          {[["day", "Day"], ["week", "Week"], ["plan", "Plan"]].map(([k, l]) => (
            <button key={k} role="tab" aria-selected={view === k} className={view === k ? "active" : ""} onClick={() => setView(k)}>{l}</button>
          ))}
        </nav>
      </header>

      <main>
        {/* ---------- DAY ---------- */}
        {view === "day" && (
          <section>
            <div className="nav-row">
              <button className="step" disabled={sel <= 1} onClick={() => setSel(sel - 1)} aria-label="Previous day">‹</button>
              <div className="nav-mid">
                <div className="kicker" style={{ color: phase.color }}>{phase.name}, Week {week.n}</div>
                <h2>Day {sel}</h2>
                <div className="date">{fmtLong(dateOf(sel))}</div>
              </div>
              <button className="step" disabled={sel >= 90} onClick={() => setSel(sel + 1)} aria-label="Next day">›</button>
            </div>

            {sel !== todayClamped && (
              <button className="link" onClick={() => setSel(todayClamped)}>Jump to today</button>
            )}

            {yesterdayMissed && (
              <div className="nudge">Yesterday was missed. Never miss twice: even the minimum day counts today.</div>
            )}

            <Field label="Today's one most important win (set it this morning)" rows={2} value={dayData.win} onChange={(v) => patchDay(sel, { win: v })} />

            <div className="card">
              <div className="card-head">
                <h3>Daily checklist</h3>
                <span className="frac" style={{ color: phase.color }}>{dayCount}/9</span>
              </div>
              <div className="bar"><div style={{ width: `${(dayCount / 9) * 100}%`, background: phase.color }} /></div>
              {DAILY.map((l, i) => (
                <CheckRow key={l} label={l} color={phase.color} checked={dayData.checks[i]} onToggle={() => toggleDay(sel, i)} />
              ))}
              <details className="min">
                <summary>Minimum viable day</summary>
                <ul>{MIN_DAY.map((m) => <li key={m}>{m}</li>)}</ul>
                <p>A minimum day is not the goal. It is the safety net that prevents one difficult day from becoming a lost week.</p>
              </details>
            </div>

            <div className="card">
              <h3>Evening review</h3>
              <Field label="Today's most important win" value={dayData.won} onChange={(v) => patchDay(sel, { won: v })} />
              <Field label="Where did I drift?" value={dayData.drift} onChange={(v) => patchDay(sel, { drift: v })} />
              <Field label="What will I do differently tomorrow?" value={dayData.tomorrow} onChange={(v) => patchDay(sel, { tomorrow: v })} />
            </div>

            <div className="pager">
              <button className="step wide" disabled={sel <= 1} onClick={() => setSel(sel - 1)}>‹ Day {sel - 1}</button>
              <button className="step wide" disabled={sel >= 90} onClick={() => setSel(sel + 1)}>Day {sel + 1} ›</button>
            </div>
          </section>
        )}

        {/* ---------- WEEK ---------- */}
        {view === "week" && (() => {
          const w = week;
          const wd = getWeek(w.n);
          const wp = phaseOf(w.from);
          const days = Array.from({ length: w.to - w.from + 1 }, (_, i) => w.from + i);
          const weekFull = days.filter((n) => count(data.days[n]) === 9).length;
          return (
            <section>
              <div className="nav-row">
                <button className="step" disabled={w.n <= 1} onClick={() => setSel(WEEKS[w.n - 2].from)} aria-label="Previous week">‹</button>
                <div className="nav-mid">
                  <div className="kicker" style={{ color: wp.color }}>{wp.name}</div>
                  <h2>Week {w.n}: {w.title}</h2>
                  <div className="date">{fmtShort(dateOf(w.from))} to {fmtShort(dateOf(w.to))}, {weekFull} of {days.length} days complete</div>
                </div>
                <button className="step" disabled={w.n >= 13} onClick={() => setSel(WEEKS[w.n].from)} aria-label="Next week">›</button>
              </div>

              <div className="week-days">
                {days.map((n) => {
                  const c = count(data.days[n]);
                  return (
                    <button key={n} className={"wd" + (n === sel ? " sel" : "")} onClick={() => goDay(n)} style={{ "--c": wp.color }}>
                      <span>{dateOf(n).toLocaleDateString(undefined, { weekday: "short" })}</span>
                      <b>{dateOf(n).getDate()}</b>
                      <small>{c}/9</small>
                    </button>
                  );
                })}
              </div>

              <div className="card">
                <h3>This week's focus</h3>
                {w.items.map((l, i) => (
                  <CheckRow key={l} label={l} color={wp.color} checked={wd.checks[i]} onToggle={() => patchWeek(w.n, { checks: setIn(wd.checks, i, !wd.checks[i]) })} />
                ))}
              </div>

              <div className="card">
                <h3>This week's top 3</h3>
                {[0, 1, 2].map((i) => (
                  <label className="top3" key={i}>
                    <span style={{ background: wp.color }}>{i + 1}</span>
                    <input value={wd.top[i]} placeholder="Write here" onChange={(e) => patchWeek(w.n, { top: setIn(wd.top, i, e.target.value) })} />
                  </label>
                ))}
              </div>

              <div className="card">
                <h3>Weekly review</h3>
                {WEEK_REVIEW_Q.map((q, i) => (
                  <Field key={q} label={q} value={wd.review[i]} onChange={(v) => patchWeek(w.n, { review: setIn(wd.review, i, v) })} />
                ))}
              </div>
            </section>
          );
        })()}

        {/* ---------- PLAN ---------- */}
        {view === "plan" && (
          <section>
            <blockquote className="verse">
              “Remember therefore from where you have fallen; repent and do the first works.”
              <cite>Revelation 2:5</cite>
            </blockquote>

            <div className="card">
              <h3>Five rules</h3>
              <ol className="rules">
                {RULES.map(([a, b]) => (
                  <li key={a}><b>{a}</b> {b}</li>
                ))}
              </ol>
            </div>

            <div className="card">
              <h3>My 90-day targets</h3>
              <p className="hint">By December 29, I want to…</p>
              {TARGETS.map(([area, def]) => (
                <Field
                  key={area}
                  label={area}
                  placeholder={def}
                  value={data.plan.targets[area]}
                  onChange={(v) => patchPlan({ targets: { ...data.plan.targets, [area]: v } })}
                />
              ))}
              <Field label="My primary 90-day skill" rows={1} value={data.plan.skill} onChange={(v) => patchPlan({ skill: v })} />
              <Field label="My 90-day output / project" rows={2} value={data.plan.output} onChange={(v) => patchPlan({ output: v })} />
              <div className="sub-label">My three biggest sources of drift to reduce</div>
              {[0, 1, 2].map((i) => (
                <label className="top3" key={i}>
                  <span style={{ background: "#464aa3" }}>{i + 1}</span>
                  <input value={data.plan.drift[i]} placeholder="Write here" onChange={(e) => patchPlan({ drift: setIn(data.plan.drift, i, e.target.value) })} />
                </label>
              ))}
            </div>

            {PHASES.map((p) => {
              const pc = data.plan.phaseChecks[p.id] || Array(5).fill(false);
              const pr = data.plan.phases[p.id] || ["", "", ""];
              return (
                <div className="card" key={p.id} style={{ borderTop: `4px solid ${p.color}` }}>
                  <h3 style={{ color: p.color }}>Phase {p.id}: {p.name}</h3>
                  <p className="hint">{p.range}, {p.focus}</p>
                  {p.items.map((l, i) => (
                    <CheckRow key={l} label={l} color={p.color} checked={pc[i]} onToggle={() => patchPlan({ phaseChecks: { ...data.plan.phaseChecks, [p.id]: setIn(pc, i, !pc[i]) } })} />
                  ))}
                  <div className="sub-label">Phase review</div>
                  {PHASE_Q.map((q, i) => (
                    <Field key={q} label={q} value={pr[i]} onChange={(v) => patchPlan({ phases: { ...data.plan.phases, [p.id]: setIn(pr, i, v) } })} />
                  ))}
                </div>
              );
            })}

            <div className="card">
              <h3>Daily operating system</h3>
              <p className="hint">Adjust the clock times to your real schedule; keep the order and priorities.</p>
              <dl className="os">
                {OS.map(([a, b]) => (
                  <div key={a}><dt>{a}</dt><dd>{b}</dd></div>
                ))}
              </dl>
            </div>
          </section>
        )}
      </main>
    </div>
  );
}
